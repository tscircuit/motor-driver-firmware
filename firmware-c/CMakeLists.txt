cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Pico SDK directory")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(motorcontrol C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
set(MOTOR_INITIAL_NAME "Motor bench" CACHE STRING "Name used when native settings are empty")
set(MOTOR_INITIAL_THRESHOLD 65 CACHE STRING "Alarm threshold used when native settings are empty")
add_executable(motorcontrol main.c motor.c profile.c settings.c current_capture.c current_stats.c usb_descriptors.c vendor/cJSON.c)
# Fingerprint the source and build configuration so live boards can be checked
# against the actual image, without relying only on a fixed version label.
set(BUILD_INPUTS current_capture.c current_capture.h current_stats.c CMakeLists.txt main.c motor.c motor.h profile.c profile.h settings.c
    settings.h usb_descriptors.c tusb_config.h stepper.pio memory_guard.ld
    vendor/cJSON.c vendor/cJSON.h)
set(BUILD_FINGERPRINT "${PICO_SDK_VERSION_STRING}:${MOTOR_INITIAL_NAME}:${MOTOR_INITIAL_THRESHOLD}")
foreach(INPUT IN LISTS BUILD_INPUTS)
    file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/${INPUT}" INPUT_HASH)
    string(APPEND BUILD_FINGERPRINT ":${INPUT_HASH}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/${INPUT}")
endforeach()
string(SHA256 BUILD_FINGERPRINT "${BUILD_FINGERPRINT}")
string(SUBSTRING "${BUILD_FINGERPRINT}" 0 16 FIRMWARE_BUILD_ID)
configure_file(build_id.h.in build_id.h @ONLY)
target_include_directories(motorcontrol PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(motorcontrol PRIVATE MOTOR_INITIAL_NAME="${MOTOR_INITIAL_NAME}" MOTOR_INITIAL_THRESHOLD=${MOTOR_INITIAL_THRESHOLD} PICO_FLASH_SIZE_BYTES=2097152 PICO_STACK_SIZE=4096 PICO_CORE1_STACK_SIZE=4096 CJSON_NESTING_LIMIT=8)
target_compile_options(motorcontrol PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
target_link_libraries(motorcontrol pico_stdlib pico_rand pico_unique_id pico_multicore pico_flash hardware_pio hardware_dma hardware_i2c hardware_adc hardware_pwm hardware_flash hardware_sync hardware_watchdog tinyusb_device tinyusb_board m)
pico_generate_pio_header(motorcontrol ${CMAKE_CURRENT_LIST_DIR}/stepper.pio)
target_link_options(motorcontrol PRIVATE "LINKER:-T,${CMAKE_CURRENT_LIST_DIR}/memory_guard.ld")
pico_set_program_name(motorcontrol "motorcontrol-native")
pico_set_program_version(motorcontrol "1.0.0")
pico_enable_stdio_usb(motorcontrol 0)
pico_enable_stdio_uart(motorcontrol 0)
pico_add_extra_outputs(motorcontrol)
